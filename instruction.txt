devices_logger.py
#Программа для логирования данных с приборов

Имена приборов и их конфигов:
    '-d', '--devices', nargs='+', type=str, required=True
    синтаксис: имя_прибора[.номер_канала]:имя_пресета [имя_прибора...]
Запуск измерений без предварительной настройки приборов:
    '-fs', '--faststart', action='store_true'
Имена каналов для вывода на график:
    '-g', '--graph', nargs='+', type=str, default=None
    синтаксис: имя_прибора.величина[номер_канала] [имя_прибора...]
Период одного шага измерения в секундах:
    '-p', '--period', type=float, default=1.0
Имя файла для сохранения результатов. Если не указано — генерируется автоматически:
    '-f', '--filename', type=str, default=None
Количество измерений (точек):
    '-n', '--points', type=int, default=999999
Количество точек по оси X на графике:
    '-cp', '--canvaspoints', type=int, default=3600

Пример:
python devices_logger.py -d TH1992B_1.1:APL_I TH1992B_1.2:APL_I TH2690A_1:FDUK -g TH1992B_1.CURR1 TH1992B_1.CURR2 TH2690A_1.CURR -n 300 -cp 200 -fs

axes_scanner.py
#Программа для автоматизации серии экспериментов сканирования щелью

Имена приборов и их конфигов:
    '-d', '--devices', nargs='+', type=str, required=True
    синтаксис: имя_прибора[.номер_канала]:имя_пресета [имя_прибора...]
    например: -d TH1992B_1.1:FDUCK_I TH1992B_1.2:APL_I TH2690A_1:PICOAMMETER_TEST_1
Индексы экспериментов для запуска (например: -r 0 2 3):
    '-r', '--run', nargs='+', type=int, default=None
    Если параметр не указан — запустятся ВСЕ эксперименты из файла
Путь к текстовому файлу с таблицей заданий:
    '-tf', '--taskfile', type=str, default='Axes_scan_default.txt'
Запуск измерений без предварительной настройки приборов:
    -fs', '--faststart', action='store_true'
Имена каналов и трансформаций для вывода на график:
    '-g', '--graph', nargs='+', type=str, default=None
    синтаксис: имя_прибора.величина[номер_канала][=величина:имя_функции] [имя_прибора...]
    например: -g TH1992B_1.CURR1 TH1992B_1.RES1=T1:Pt100_default
Сохранять результаты всех экспериментов в один файл:
    '-sf', '--singlefile', action='store_true'
Ось X для графика:
    '-x', '--xaxis', type=str, default='time', choices=['time', 'APT', 'APL', 'APR', 'APB']

Пример:
python axes_scanner.py -d TH1992B_1.1:APL_I TH1992B_1.2:APL_I -g TH1992B_1.CURR1 TH1992B_1.CURR2 -tf EXP2_APL+APT.txt

Конфигурация TH1992B

#Режим работы источника
#VOLT, CURR
Mode = VOLT
#Режим сигнала источника
#FIX (DC), SWE (SWEep), LIST (по точкам)
AWG = FIX
#Диапазон сигнала источника напр./тока
#для VOLT - 0.2, 2, 20, 200 (V)
#для CURR - 10e-9, 100e-9, 1e-6, 10e-6... 1, 1.5, 3 (A)
SOURCE-range = 200
#Режим автоматического переключения диапазонов измерения
#ON, OFF
SENSE-range auto = OFF
#Диапазон измерения
#для VOLT - 0.2, 2, 20, 200 (V)
#для CURR - 10e-9, 100e-9, 1e-6, 10e-6... 1, 1.5, 3 (A)
SENSE-range = 10e-06
#Лимит измерения напр./тока (V или A)
SENSE Limit = 100e-03
#Значение сигнала напр./тока источника (V или A)
SOURCE-value = -24
#Способ отключения канала
#ZERO, HIZ, NORM
Off state = HIZ
#Значение апертуры измерения
#это время, за которое производится усреднение (s)
Aperture = 0.02
#Режим заземления клеммы Low
#GRO (GROund, замыкание на корпус), FLO (FLOat, не заземлено)
Low T = GRO
#Overvoltage/overcurrent protection
#1, 0
OVP/OCP = 0
#Вкл/выкл фильтра выходного сигнала
#1, 0
Filter = 1
#Time constant of output filter (s)
#LPF-t = 5e-6
#High capacitance mode
#1, 0
High C = 0
#Remote function (4-wire)
#1 (используется схема подключения 4-wire), 0 (2-wire)
Sensing = 0
